export default function definePacket<T>(
    name: string,
    codecOrOptions?: Type.Codec.External<T> | Type.Packet.Options,
    options?: Type.Packet.Options,
) {
    let codec: Type.Codec.Internal<T> | undefined;
    let opts: Type.Packet.Options | undefined;

    if (codecOrOptions !== undefined && "write" in (codecOrOptions as object)) {
        codec = codecOrOptions as Type.Codec.Internal<T>;
        opts = options;
    } else {
        codec = undefined;
        opts = codecOrOptions as Type.Packet.Options | undefined;
    }

    const unreliable = opts?.unreliable ?? false;

    const def = Registry.register(
        name,
        codec as Type.Codec.Internal<unknown> | undefined,
        unreliable,
    );

    Report.log("debug", "PACKET_DEFINED", { name, unreliable });

    function send(dataOrTarget?: T | Type.Target, target?: Type.Target) {
        if (def.id === 0) {
            Report.log("warn", "PACKET_SEND_BEFORE_START", { name });
            return;
        }

        const resolvedData = codec !== undefined ? (dataOrTarget as T) : undefined;
        const resolvedTarget =
            codec !== undefined ? target : (dataOrTarget as Type.Target | undefined);

        Outbound.send(def.id, codec, resolvedData, resolvedTarget, unreliable);
    }

    function on(handler: (player: Player, data: T) => void) {
        Registry.setHandler(name, handler as (player: Player, data: unknown) => void);
    }

    return { send, on } as Type.Packet.Definition<T>;
}

// Internal
import Constant from "@constant";

// Security
import Report from "@security/report";

const INITIAL_SIZE = 1024;
const MAX_SIZE = Constant.MAX_PAYLOAD * 2;

export class Channel {
    public buf: buffer;
    public offset: number = 0;
    public size: number;

    public deltas: Map<number, unknown> = new Map();
    public prevHash: number = 0;

    private readonly f32scratch: buffer;

    constructor(initialSize: number = INITIAL_SIZE) {
        this.size = initialSize;
        this.buf = buffer.create(initialSize);
        this.f32scratch = buffer.create(4);
    }

    public reset(): void {
        this.offset = 0;
    }

    public used(): number {
        return this.offset;
    }

    private grow(needed: number): void {
        let newSize = this.size;
        while (newSize < this.offset + needed) {
            newSize *= 2;
            if (newSize > MAX_SIZE) {
                Report.log("fatal", "CHANNEL_OVERFLOW", {
                    offset: this.offset,
                    needed,
                    size: newSize,
                });
            }
        }
        const newBuf = buffer.create(newSize);
        buffer.copy(newBuf, 0, this.buf, 0, this.offset);
        this.buf = newBuf;
        this.size = newSize;
    }

    public ensureSpace(bytes: number): void {
        if (this.offset + bytes > this.size) this.grow(bytes);
    }

    public copyOut(s: number, e: number): buffer {
        const size = e - s;
        const out = buffer.create(size);
        buffer.copy(out, 0, this.buf, s, size);
        return out;
    }

    public varint(value: number): void {
        while (value >= 0x80) {
            this.u8(bit32.bor(bit32.band(value, 0x7f), 0x80));
            value = math.floor(value / 128);
        }
        this.u8(value);
    }

    public u8(value: number): void {
        this.ensureSpace(1);
        buffer.writeu8(this.buf, this.offset, value);
        this.offset += 1;
    }

    public u16(value: number): void {
        this.ensureSpace(2);
        buffer.writeu16(this.buf, this.offset, value);
        this.offset += 2;
    }

    public u24(value: number): void {
        this.ensureSpace(3);
        buffer.writeu8(this.buf, this.offset, bit32.band(value, 0xff));
        buffer.writeu8(this.buf, this.offset + 1, bit32.band(bit32.rshift(value, 8), 0xff));
        buffer.writeu8(this.buf, this.offset + 2, bit32.band(bit32.rshift(value, 16), 0xff));
        this.offset += 3;
    }

    public u32(value: number): void {
        this.ensureSpace(4);
        buffer.writeu32(this.buf, this.offset, value);
        this.offset += 4;
    }

    public i8(value: number): void {
        this.ensureSpace(1);
        buffer.writei8(this.buf, this.offset, value);
        this.offset += 1;
    }

    public i16(value: number): void {
        this.ensureSpace(2);
        buffer.writei16(this.buf, this.offset, value);
        this.offset += 2;
    }

    public i32(value: number): void {
        this.ensureSpace(4);
        buffer.writei32(this.buf, this.offset, value);
        this.offset += 4;
    }

    public f16(value: number): void {
        this.ensureSpace(2);

        if (value !== value) {
            buffer.writeu16(this.buf, this.offset, 0x7e00);
            this.offset += 2;
            return;
        }

        const sign = value < 0 ? 1 : 0;
        value = math.abs(value);

        if (value === math.huge) {
            buffer.writeu16(this.buf, this.offset, sign * 0x8000 + 0x7c00);
            this.offset += 2;
            return;
        }

        if (value === 0) {
            buffer.writeu16(this.buf, this.offset, 0);
            this.offset += 2;
            return;
        }

        buffer.writef32(this.f32scratch, 0, value);
        const bits = buffer.readu32(this.f32scratch, 0);

        const exp = bit32.band(bit32.rshift(bits, 23), 0xff) - 127 + 15;
        const mantissa = bit32.band(bits, 0x7fffff);

        if (exp >= 31) {
            buffer.writeu16(this.buf, this.offset, bit32.bor(bit32.lshift(sign, 15), 0x7c00));
        } else if (exp <= 0) {
            buffer.writeu16(this.buf, this.offset, 0);
        } else {
            buffer.writeu16(
                this.buf,
                this.offset,
                bit32.bor(
                    bit32.lshift(sign, 15),
                    bit32.lshift(exp, 10),
                    bit32.rshift(mantissa, 13),
                ),
            );
        }

        this.offset += 2;
    }

    public f32(value: number): void {
        this.ensureSpace(4);
        buffer.writef32(this.buf, this.offset, value);
        this.offset += 4;
    }

    public f64(value: number): void {
        this.ensureSpace(8);
        buffer.writef64(this.buf, this.offset, value);
        this.offset += 8;
    }

    public bool(value: boolean): void {
        this.u8(value ? 1 : 0);
    }

    public bitArray(values: boolean[], len: number): void {
        const byteCount = math.ceil(len / 8);
        this.ensureSpace(byteCount);

        const buf = this.buf;
        let off = this.offset;
        let byte = 0;
        let bit = 0;

        for (const v of values) {
            if (v) byte = bit32.bor(byte, bit32.lshift(1, bit));
            bit++;
            if (bit === 8) {
                buffer.writeu8(buf, off++, byte);
                byte = 0;
                bit = 0;
            }
        }

        if (bit > 0) buffer.writeu8(buf, off, byte);
        this.offset += byteCount;
    }

    public string(value: string, len: number): void {
        this.ensureSpace(len);
        buffer.writestring(this.buf, this.offset, value);
        this.offset += len;
    }

    public bytes(src: buffer, srcOffset: number, length: number): void {
        this.ensureSpace(length);
        buffer.copy(this.buf, this.offset, src, srcOffset, length);
        this.offset += length;
    }
}

export default Channel;

export class Cursor {
    private player: Player | undefined;

    public buf: buffer;
    public offset = 0;

    constructor(buf: buffer, player?: Player) {
        this.buf = buf;
        this.player = player;
    }

    public reset(buf: buffer) {
        this.buf = buf;
        this.offset = 0;
    }

    public remaining(): number {
        return buffer.len(this.buf) - this.offset;
    }

    public ensureRemaining(bytes: number) {
        if (this.offset + bytes > buffer.len(this.buf)) {
            if (IS_SERVER && this.player !== undefined) {
                Report.log("kick", "CURSOR_UNDERFLOW", {
                    player: this.player,
                    offset: this.offset,
                    needed: bytes,
                    size: buffer.len(this.buf),
                });
            } else {
                Report.log("warn", "CURSOR_UNDERFLOW", {
                    offset: this.offset,
                    needed: bytes,
                    size: buffer.len(this.buf),
                });
            }
        }
    }

    public varint(): number {
        let result = 0;
        let shift = 0;
        while (true) {
            const byte = this.u8();
            result = bit32.bor(result, bit32.lshift(bit32.band(byte, 0x7f), shift));
            if (bit32.band(byte, 0x80) === 0) return result;
            shift += 7;
            if (shift > 28) {
                Report.log("warn", "VARINT_TOO_LONG");
                return result;
            }
        }
    }

    public u8(): number {
        this.ensureRemaining(1);
        const v = buffer.readu8(this.buf, this.offset);
        this.offset += 1;
        return v;
    }

    public u16(): number {
        this.ensureRemaining(2);
        const v = buffer.readu16(this.buf, this.offset);
        this.offset += 2;
        return v;
    }

    public u24(): number {
        this.ensureRemaining(3);
        const v = bit32.bor(
            buffer.readu8(this.buf, this.offset),
            bit32.lshift(buffer.readu8(this.buf, this.offset + 1), 8),
            bit32.lshift(buffer.readu8(this.buf, this.offset + 2), 16),
        );
        this.offset += 3;
        return v;
    }

    public u32(): number {
        this.ensureRemaining(4);
        const v = buffer.readu32(this.buf, this.offset);
        this.offset += 4;
        return v;
    }

    public i8(): number {
        this.ensureRemaining(1);
        const v = buffer.readi8(this.buf, this.offset);
        this.offset += 1;
        return v;
    }

    public i16(): number {
        this.ensureRemaining(2);
        const v = buffer.readi16(this.buf, this.offset);
        this.offset += 2;
        return v;
    }

    public i32(): number {
        this.ensureRemaining(4);
        const v = buffer.readi32(this.buf, this.offset);
        this.offset += 4;
        return v;
    }

    public f16(): number {
        this.ensureRemaining(2);
        const bits = buffer.readu16(this.buf, this.offset);
        this.offset += 2;

        const sign = bit32.rshift(bits, 15) === 1 ? -1 : 1;
        const exp = bit32.band(bit32.rshift(bits, 10), 0x1f);
        const mantissa = bit32.band(bits, 0x3ff);

        if (exp === 0x1f) {
            return mantissa === 0 ? sign * math.huge : 0 / 0;
        }

        if (exp === 0) {
            return sign * math.pow(2, -14) * (mantissa / 1024);
        }

        return sign * math.pow(2, exp - 15) * (1 + mantissa / 1024);
    }

    public f32(): number {
        this.ensureRemaining(4);
        const v = buffer.readf32(this.buf, this.offset);
        this.offset += 4;
        return v;
    }

    public f64(): number {
        this.ensureRemaining(8);
        const v = buffer.readf64(this.buf, this.offset);
        this.offset += 8;
        return v;
    }

    public bool(): boolean {
        return this.u8() !== 0;
    }

    public bitArray(len: number): boolean[] {
        const byteCount = math.ceil(len / 8);
        this.ensureRemaining(byteCount);
        const result = new Array<boolean>(len);
        for (let i = 0; i < len; i++) {
            const byteIdx = math.floor(i / 8);
            const bit = i % 8;
            result[i] =
                bit32.band(buffer.readu8(this.buf, this.offset + byteIdx), bit32.lshift(1, bit)) !==
                0;
        }
        this.offset += byteCount;
        return result;
    }

    public string(len: number): string {
        this.ensureRemaining(len);
        const v = buffer.readstring(this.buf, this.offset, len);
        this.offset += len;
        return v;
    }

    public copyRange(s: number, e: number): buffer {
        const size = e - s;
        const out = buffer.create(size);
        buffer.copy(out, 0, this.buf, s, size);
        return out;
    }

    public bytes(length: number): buffer {
        this.ensureRemaining(length);
        const out = buffer.create(length);
        buffer.copy(out, 0, this.buf, this.offset, length);
        this.offset += length;
        return out;
    }

    public getPlayer(): Player | undefined {
        return this.player;
    }
}

// Internal
import Constant from "@constant";
import * as Type from "@type";

// Security
import Report from "@security/report";

function array<T>(codec: Type.Codec.External<T>) {
    const internalCodec = codec as Type.Codec.Internal<T>;
    const isBool = typeOf(internalCodec._default) === "boolean";
    const elemSize = internalCodec._size;

    const directWrite = internalCodec._directWrite;
    const directRead = internalCodec._directRead;
    const isDirect =
        elemSize !== undefined && directWrite !== undefined && directRead !== undefined;

    const write = internalCodec.write;
    const read = internalCodec.read;

    return {
        write: (ch, value) => {
            const len = value.size();
            if (len > Constant.MAX_ARRAY_LENGTH) {
                Report.log("warn", "ARRAY_TOO_LARGE", {
                    length: len,
                    max: Constant.MAX_ARRAY_LENGTH,
                });
            }
            ch.varint(len);

            if (isBool) {
                ch.bitArray(value as boolean[], len);
            } else if (isDirect) {
                const payloadBytes = len * elemSize;
                ch.ensureSpace(payloadBytes);
                const buf = ch.buf;
                let off = ch.offset;
                for (const v of value) {
                    directWrite(buf, off, v);
                    off += elemSize;
                }
                ch.offset += payloadBytes;
            } else {
                for (const v of value) write(ch, v);
            }
        },
        read: (cursor) => {
            const len = cursor.varint();
            if (len > Constant.MAX_ARRAY_LENGTH) {
                Report.log("warn", "ARRAY_TOO_LARGE", { len, max: Constant.MAX_ARRAY_LENGTH });
            }

            if (isBool) return cursor.bitArray(len) as unknown as T[];

            if (isDirect) {
                const payloadBytes = len * elemSize;
                cursor.ensureRemaining(payloadBytes);

                const result = new Array<T>(len);
                const buf = cursor.buf;

                let off = cursor.offset;
                for (let i = 0; i < len; i++) {
                    result[i] = directRead(buf, off) as T;
                    off += elemSize;
                }
                cursor.offset += payloadBytes;

                return result;
            }

            const result = new Array<T>(len);
            for (let i = 0; i < len; i++) {
                result[i] = read(cursor);
            }
            return result;
        },
        _default: [] as T[],
    } as Type.Codec.Internal<T[]> as Type.Codec.External<T[]>;
}

export default array;

// Internal
import * as Type from "@type";

// Binary
import Channel from "@binary/channel";

// Transport — only for allocId(); write-path cache is now ch.deltas
import Snapshot from "@transport/snapshot";

// Security
import Report from "@security/report";

const FLAG_UNCHANGED = 0;
const FLAG_FULL = 1;
const FLAG_DIFF = 2;

// Bool bitset diff flag — byte-level sparse patch of the packed bitset.
// Wire format: FLAG_BITSET_DIFF + changedByteCount(varint) + [byteIndex(varint) + byteValue(u8)]…
const FLAG_BITSET_DIFF = 3;

interface CachedElement<T> {
    value: T;
    bytes: buffer;
}

interface CachedBoolArray {
    // Packed bitset representation of the last-sent bool array.
    // Length = ceil(len / 8) bytes.
    bitset: buffer;
    len: number;
}

function varIntLength(n: number): number {
    if (n < 0x80) return 1;
    if (n < 0x4000) return 2;
    if (n < 0x200000) return 3;
    if (n < 0x10000000) return 4;
    return 5;
}

// Serializes a single element into the scratch channel.
// Returns [startOffset, endOffset] — does NOT allocate a new buffer.
function serializeElementInPlace<T>(
    scratch: Channel,
    write: (ch: Channel, value: T) => void,
    value: T,
): void {
    scratch.offset = 0;
    write(scratch, value);
}

// Pack a bool array into a fresh buffer (byteCount = ceil(len/8)).
function packBitset(value: boolean[], len: number): buffer {
    const byteCount = math.ceil(len / 8);
    const out = buffer.create(byteCount);
    for (let i = 0; i < len; i++) {
        if (value[i]) {
            const byteIdx = math.floor(i / 8);
            const bit = i % 8;
            buffer.writeu8(
                out,
                byteIdx,
                bit32.bor(buffer.readu8(out, byteIdx), bit32.lshift(1, bit)),
            );
        }
    }
    return out;
}

// Write a full bitset to the channel (no length prefix — caller sends varint(len) separately).
function writeBitset(ch: Channel, bs: buffer, byteCount: number): void {
    ch.bytes(bs, 0, byteCount);
}

// ─── Bool fast path ───────────────────────────────────────────────────────────
//
// Instead of encoding each bool as a separate 1-byte element and doing
// element-level diff (FLAG_DIFF with index varints), we:
//   1. Pack all bools into a bitset buffer (125 bytes for 1000 bools).
//   2. XOR old vs new bitsets to find changed bytes.
//   3. If #changed_bytes * ~3 < byteCount → send FLAG_BITSET_DIFF + sparse patches.
//   4. Otherwise send FLAG_FULL + the whole bitset.
//
// For 1 flip in 1000 bools: FLAG_BITSET_DIFF(1) + count(1) + byteIdx(1-2) + val(1) = 4-5 bytes.
// vs the old element-sparse approach: flag(1) + count(1) + 1000 * (index_varint + value_byte).

function writeBoolDelta(
    ch: Channel,
    value: boolean[],
    cached: CachedBoolArray | undefined,
    newLen: number,
): CachedBoolArray {
    const byteCount = math.ceil(newLen / 8);
    const newBitset = packBitset(value, newLen);

    if (cached === undefined || cached.len !== newLen) {
        // Full send
        ch.u8(FLAG_FULL);
        ch.varint(newLen);
        writeBitset(ch, newBitset, byteCount);
        return { bitset: newBitset, len: newLen };
    }

    // Find changed bytes via XOR
    const changedIndices: number[] = [];
    for (let i = 0; i < byteCount; i++) {
        if (buffer.readu8(newBitset, i) !== buffer.readu8(cached.bitset, i)) {
            changedIndices.push(i);
        }
    }

    if (changedIndices.size() === 0) {
        ch.u8(FLAG_UNCHANGED);
        return cached;
    }

    // Cost comparison:
    //   DIFF cost  = 1(flag) + varint(count) + count*(varint(idx) + 1)
    //   FULL cost  = 1(flag) + varint(len) + byteCount
    let diffCost = 1 + varIntLength(changedIndices.size());
    for (const idx of changedIndices) diffCost += varIntLength(idx) + 1;
    const fullCost = 1 + varIntLength(newLen) + byteCount;

    if (diffCost < fullCost) {
        ch.u8(FLAG_BITSET_DIFF);
        ch.varint(changedIndices.size());
        for (const idx of changedIndices) {
            ch.varint(idx);
            ch.u8(buffer.readu8(newBitset, idx));
        }
    } else {
        ch.u8(FLAG_FULL);
        ch.varint(newLen);
        writeBitset(ch, newBitset, byteCount);
    }

    return { bitset: newBitset, len: newLen };
}

function readBoolDelta(cursor: Type.Cursor, cached: CachedBoolArray | undefined): boolean[] {
    const flag = cursor.u8();

    if (flag === FLAG_UNCHANGED) {
        if (cached === undefined) {
            Report.log("warn", "DELTA_ARRAY_BOOL_UNCHANGED_NO_CACHE", {});
            return [];
        }
        // Unpack from cached bitset
        const result = new Array<boolean>(cached.len);
        for (let i = 0; i < cached.len; i++) {
            const byteIdx = math.floor(i / 8);
            const bit = i % 8;
            result[i] =
                bit32.band(buffer.readu8(cached.bitset, byteIdx), bit32.lshift(1, bit)) !== 0;
        }
        return result;
    }

    if (flag === FLAG_FULL) {
        const len = cursor.varint();
        const byteCount = math.ceil(len / 8);
        const bs = buffer.create(byteCount);
        for (let i = 0; i < byteCount; i++) {
            buffer.writeu8(bs, i, cursor.u8());
        }
        // Unpack
        const result = new Array<boolean>(len);
        for (let i = 0; i < len; i++) {
            const byteIdx = math.floor(i / 8);
            const bit = i % 8;
            result[i] = bit32.band(buffer.readu8(bs, byteIdx), bit32.lshift(1, bit)) !== 0;
        }
        return result;
    }

    // FLAG_BITSET_DIFF
    if (cached === undefined) {
        Report.log("warn", "DELTA_ARRAY_BOOL_DIFF_NO_CACHE", {});
        const count = cursor.varint();
        for (let i = 0; i < count; i++) {
            cursor.varint(); // byteIdx
            cursor.u8(); // byteVal
        }
        return [];
    }

    const count = cursor.varint();
    // Mutate a copy of the bitset
    const newBs = buffer.create(buffer.len(cached.bitset));
    buffer.copy(newBs, 0, cached.bitset, 0, buffer.len(cached.bitset));
    for (let i = 0; i < count; i++) {
        const byteIdx = cursor.varint();
        const byteVal = cursor.u8();
        buffer.writeu8(newBs, byteIdx, byteVal);
    }
    const len = cached.len;
    const result = new Array<boolean>(len);
    for (let i = 0; i < len; i++) {
        const byteIdx = math.floor(i / 8);
        const bit = i % 8;
        result[i] = bit32.band(buffer.readu8(newBs, byteIdx), bit32.lshift(1, bit)) !== 0;
    }
    return result;
}

// ─── Generic element delta ────────────────────────────────────────────────────

function deltaArray<T>(codec: Type.Codec.External<T>) {
    const id = Snapshot.allocId();
    const internalCodec = codec as unknown as Type.Codec.Internal<T>;

    // Detect bool codec: _default is boolean and _size is 1.
    const isBool = typeOf(internalCodec._default) === "boolean";

    // Bool arrays get their own id for the bitset cache.
    // Generic arrays reuse `id` for the CachedElement[] cache.

    // Scratch channel: used only for per-element serialization to produce
    // comparison buffers. Never connected to a player's outbound channel.
    const scratch = new Channel();

    const write = internalCodec.write;
    const read = internalCodec.read;

    return {
        write: (ch, value) => {
            const len = value.size();

            // ── Bool fast path ──────────────────────────────────────────────
            if (isBool) {
                const cached = ch.deltas.get(id) as CachedBoolArray | undefined;
                const newCache = writeBoolDelta(ch, value as unknown as boolean[], cached, len);
                ch.deltas.set(id, newCache);
                return;
            }

            // ── Generic path ────────────────────────────────────────────────
            const cached = ch.deltas.get(id) as CachedElement<T>[] | undefined;

            if (cached === undefined || len !== cached.size()) {
                ch.u8(FLAG_FULL);
                ch.varint(len);
                const newCache: CachedElement<T>[] = [];
                for (let i = 0; i < len; i++) {
                    serializeElementInPlace(scratch, write, value[i]);
                    const bytes = scratch.copyOut(0, scratch.offset);
                    newCache[i] = { value: value[i], bytes };
                    ch.bytes(bytes, 0, buffer.len(bytes));
                }
                ch.deltas.set(id, newCache);
                return;
            }

            // One pass: skip re-serialization when value reference is identical,
            // otherwise serialize into scratch and compare.
            const serialized: buffer[] = [];
            const dirty: number[] = [];

            for (let i = 0; i < len; i++) {
                // ── Reference equality short-circuit ──
                // If the same object reference is present, bytes cannot have changed.
                // Reuse cached bytes without touching the scratch buffer.
                if (value[i] === cached[i].value) {
                    serialized[i] = cached[i].bytes;
                    continue;
                }

                // Serialize and compare.
                serializeElementInPlace(scratch, write, value[i]);
                const elemLen = scratch.offset;

                // Compare directly against cached bytes without allocating
                // unless we actually need to store new bytes.
                const cachedBytes = cached[i].bytes;
                const cachedLen = buffer.len(cachedBytes);
                let changed = cachedLen !== elemLen;
                if (!changed) {
                    for (let j = 0; j < elemLen; j++) {
                        if (buffer.readu8(scratch.buf, j) !== buffer.readu8(cachedBytes, j)) {
                            changed = true;
                            break;
                        }
                    }
                }

                if (changed) {
                    // Allocate only when we know we need new bytes.
                    const bytes = scratch.copyOut(0, elemLen);
                    serialized[i] = bytes;
                    dirty.push(i);
                } else {
                    serialized[i] = cachedBytes;
                }
            }

            if (dirty.size() === 0) {
                ch.u8(FLAG_UNCHANGED);
                return;
            }

            // Pick cheaper encoding: DIFF (sparse) vs FULL (all).
            let fullCost = varIntLength(len);
            for (let i = 0; i < len; i++) fullCost += buffer.len(serialized[i]);

            let diffCost = varIntLength(dirty.size());
            for (const i of dirty) diffCost += varIntLength(i) + buffer.len(serialized[i]);

            if (diffCost < fullCost) {
                ch.u8(FLAG_DIFF);
                ch.varint(dirty.size());
                for (const i of dirty) {
                    ch.varint(i);
                    const b = serialized[i];
                    ch.bytes(b, 0, buffer.len(b));
                    cached[i] = { value: value[i], bytes: b };
                }
            } else {
                ch.u8(FLAG_FULL);
                ch.varint(len);
                for (let i = 0; i < len; i++) {
                    const b = serialized[i];
                    ch.bytes(b, 0, buffer.len(b));
                    cached[i] = { value: value[i], bytes: b };
                }
            }
        },

        read: (cursor) => {
            const player = cursor.getPlayer();

            // ── Bool fast path ──────────────────────────────────────────────
            if (isBool) {
                const cached = Snapshot.getCache(id, player) as CachedBoolArray | undefined;
                // Peek the flag without advancing (readBoolDelta reads it internally).
                const result = readBoolDelta(cursor, cached);
                // Rebuild bitset cache from result to keep it consistent.
                const newLen = result.size();
                const newBs = packBitset(result, newLen);
                Snapshot.setCache(id, { bitset: newBs, len: newLen } as CachedBoolArray, player);
                return result as unknown as T[];
            }

            // ── Generic path ────────────────────────────────────────────────
            const flag = cursor.u8();

            if (flag === FLAG_UNCHANGED) {
                const cached = Snapshot.getCache(id, player) as CachedElement<T>[] | undefined;
                if (cached === undefined) {
                    Report.log("warn", "DELTA_ARRAY_UNCHANGED_NO_CACHE", { id });
                    return [] as T[];
                }
                const result: T[] = [];
                for (const entry of cached) result[result.size()] = entry.value;
                return result;
            }

            if (flag === FLAG_FULL) {
                const len = cursor.varint();
                const newCache: CachedElement<T>[] = [];
                const result: T[] = [];
                for (let i = 0; i < len; i++) {
                    const start = cursor.offset;
                    const value = read(cursor);
                    const bytes = cursor.copyRange(start, cursor.offset);
                    newCache[i] = { value, bytes };
                    result[i] = value;
                }
                Snapshot.setCache(id, newCache, player);
                return result;
            }

            const cachedRaw = Snapshot.getCache(id, player) as CachedElement<T>[] | undefined;
            if (cachedRaw === undefined) {
                Report.log("warn", "DELTA_ARRAY_DIFF_NO_CACHE", { id });
                return [] as T[];
            }

            const count = cursor.varint();
            for (let i = 0; i < count; i++) {
                const index = cursor.varint();
                const start = cursor.offset;
                const value = read(cursor);
                const bytes = cursor.copyRange(start, cursor.offset);
                cachedRaw[index] = { value, bytes };
            }
            Snapshot.setCache(id, cachedRaw, player);

            const result: T[] = [];
            for (const entry of cachedRaw) result[result.size()] = entry.value;
            return result;
        },

        _default: [] as T[],
        _delta: true,
    } as Type.Codec.Internal<T[]> as Type.Codec.External<T[]>;
}

function extractQuaternion(cf: CFrame): [number, number, number, number] {
    const [_, __, ___, r00, r01, r02, r10, r11, r12, r20, r21, r22] = cf.GetComponents();
    const trace = r00 + r11 + r22;
    let qx: number, qy: number, qz: number, qw: number;

    if (trace > 0) {
        const s = 0.5 / math.sqrt(trace + 1);
        qw = 0.25 / s;
        qx = (r21 - r12) * s;
        qy = (r02 - r20) * s;
        qz = (r10 - r01) * s;
    } else if (r00 > r11 && r00 > r22) {
        const s = 2 * math.sqrt(1 + r00 - r11 - r22);
        qw = (r21 - r12) / s;
        qx = 0.25 * s;
        qy = (r01 + r10) / s;
        qz = (r02 + r20) / s;
    } else if (r11 > r22) {
        const s = 2 * math.sqrt(1 + r11 - r00 - r22);
        qw = (r02 - r20) / s;
        qx = (r01 + r10) / s;
        qy = 0.25 * s;
        qz = (r12 + r21) / s;
    } else {
        const s = 2 * math.sqrt(1 + r22 - r00 - r11);
        qw = (r10 - r01) / s;
        qx = (r02 + r20) / s;
        qy = (r12 + r21) / s;
        qz = 0.25 * s;
    }

    return [qx, qy, qz, qw];
}

function deltaCFrame(min: number, max: number, precision: number): Type.Codec.External<CFrame> {
    const id = Snapshot.allocId();
    const axis = quantizedAxis(min, max, precision);

    const cp = axis as Type.Codec.Internal<number>;

    const writePos = (ch: Type.Channel, px: number, py: number, pz: number) => {
        cp.write(ch, px);
        cp.write(ch, py);
        cp.write(ch, pz);
    };
    const readPos = (cursor: Type.Cursor) => ({
        x: cp.read(cursor),
        y: cp.read(cursor),
        z: cp.read(cursor),
    });

    const writeRot = (ch: Type.Channel, components: number[], largestIndex: number) => {
        const sign = components[largestIndex] < 0 ? -1 : 1;
        let packed = largestIndex;
        let shift = 2;
        for (let i = 0; i < 4; i++) {
            if (i === largestIndex) continue;
            const v = components[i] * sign;
            const q = math.clamp(
                math.round((v + Constant.QUAT_OFFSET) * Constant.QUAT_SCALE),
                0,
                1023,
            );
            packed = bit32.bor(packed, bit32.lshift(q, shift));
            shift += 10;
        }
        ch.u32(packed);
    };

    const readRot = (cursor: Type.Cursor) => {
        const packed = cursor.u32();
        const largestIndex = bit32.band(packed, 0x3);
        const values: number[] = [];
        for (let i = 0; i < 3; i++) {
            const q = bit32.band(bit32.rshift(packed, 2 + i * 10), 0x3ff);
            values[i] = q / Constant.QUAT_SCALE - Constant.QUAT_OFFSET;
        }
        const sumSq = values[0] ** 2 + values[1] ** 2 + values[2] ** 2;
        const largest = math.sqrt(math.max(0, 1 - sumSq));
        const q: number[] = [];
        let vi = 0;
        for (let i = 0; i < 4; i++) {
            q[i] = i === largestIndex ? largest : values[vi++];
        }
        return { largestIndex, q };
    };

    return {
        write: (ch, value) => {
            const cached = ch.deltas.get(id) as CachedCFrame | undefined;

            const pos = value.Position;
            const [qx, qy, qz, qw] = extractQuaternion(value);
            const components = [qx, qy, qz, qw];

            let largestIndex = 0;
            let largestValue = math.abs(components[0]);
            for (let i = 1; i < 4; i++) {
                if (math.abs(components[i]) > largestValue) {
                    largestValue = math.abs(components[i]);
                    largestIndex = i;
                }
            }

            const q0 = components[(largestIndex + 1) % 4];
            const q1 = components[(largestIndex + 2) % 4];
            const q2 = components[(largestIndex + 3) % 4];

            if (cached === undefined) {
                ch.u8(FLAG_FULL);
                writePos(ch, pos.X, pos.Y, pos.Z);
                writeRot(ch, components, largestIndex);
                ch.deltas.set(id, {
                    px: pos.X,
                    py: pos.Y,
                    pz: pos.Z,
                    qi: largestIndex,
                    qx: q0,
                    qy: q1,
                    qz: q2,
                });
                return;
            }

            const posDirty = cached.px !== pos.X || cached.py !== pos.Y || cached.pz !== pos.Z;
            const rotDirty =
                cached.qi !== largestIndex ||
                cached.qx !== q0 ||
                cached.qy !== q1 ||
                cached.qz !== q2;

            if (!posDirty && !rotDirty) {
                ch.u8(FLAG_UNCHANGED);
                return;
            }

            if (posDirty && rotDirty) {
                ch.u8(FLAG_FULL);
                writePos(ch, pos.X, pos.Y, pos.Z);
                writeRot(ch, components, largestIndex);
                cached.px = pos.X;
                cached.py = pos.Y;
                cached.pz = pos.Z;
                cached.qi = largestIndex;
                cached.qx = q0;
                cached.qy = q1;
                cached.qz = q2;
                return;
            }

            const dirty = (posDirty ? POS_BIT : 0) | (rotDirty ? ROT_BIT : 0);
            ch.u8(FLAG_DIFF);
            ch.u8(dirty);

            if (posDirty) {
                writePos(ch, pos.X, pos.Y, pos.Z);
                cached.px = pos.X;
                cached.py = pos.Y;
                cached.pz = pos.Z;
            }
            if (rotDirty) {
                writeRot(ch, components, largestIndex);
                cached.qi = largestIndex;
                cached.qx = q0;
                cached.qy = q1;
                cached.qz = q2;
            }
        },

        read: (cursor) => {
            const player = cursor.getPlayer();
            const flag = cursor.u8();

            if (flag === FLAG_UNCHANGED) {
                const cached = Snapshot.getCache(id, player) as CFrame | undefined;
                if (cached === undefined) {
                    Report.log("warn", "DELTA_CFRAME_UNCHANGED_NO_CACHE", { id });
                    return new CFrame();
                }
                return cached;
            }

            if (flag === FLAG_FULL) {
                const p = readPos(cursor);
                const { q } = readRot(cursor);
                const cf = new CFrame(
                    p.x,
                    p.y,
                    p.z,
                    1 - 2 * (q[1] ** 2 + q[2] ** 2),
                    2 * (q[0] * q[1] - q[2] * q[3]),
                    2 * (q[0] * q[2] + q[1] * q[3]),
                    2 * (q[0] * q[1] + q[2] * q[3]),

                );
                Snapshot.setCache(id, cf, player);

            if (cached === undefined) {

                    readPos(cursor);
                }
                if (bit32.band(dirty, ROT_BIT) !== 0) {

                }
                return new CFrame();
            }

            const dirty = cursor.u8();
            const pos =
                bit32.band(dirty, POS_BIT) !== 0
                    ? readPos(cursor)
                    : { x: cached.Position.X, y: cached.Position.Y, z: cached.Position.Z };

            const [qx, qy, qz, qw] =
                bit32.band(dirty, ROT_BIT) !== 0
                    ? (() => {
                        const { q } = readRot(cursor);
                        return q as [number, number, number, number];
                    })()
                    : extractQuaternion(cached);

            const q = [qx, qy, qz, qw];
            const cf = new CFrame(
                pos.x,
                pos.y,
                pos.z,
                1 - 2 * (q[1] ** 2 + q[2] ** 2),

            return cf;
        },

function encodeZigzag(n: number) {
	return bit32.bxor(bit32.lshift(n, 1), bit32.arshift(n, 31));
}
function decodeZigzag(n: number) {
	return bit32.bxor(bit32.rshift(n, 1), -bit32.band(n, 1));
}

function deltaInt(min: number, max: number): Type.Codec.External<number> {
	const id = Snapshot.allocId();

	return {
		write: (ch, value) => {
			const clamped = math.clamp(value, min, max);
			const prev = (ch.deltas.get(id) as number | undefined) ?? min;
			ch.varint(encodeZigzag(clamped - prev));
			ch.deltas.set(id, clamped);
		},
		read: (cursor) => {
			const player = cursor.getPlayer();
			const prev = (Snapshot.getCache(id, player) as number | undefined) ?? min;
			const value = prev + decodeZigzag(cursor.varint());
			Snapshot.setCache(id, value, player);
			return value;
		},
		_default: min,
		_delta: true,
	} as Type.Codec.Internal<number> as Type.Codec.External<number>;
}

// Internal
import * as Type from "@type";

// Binary
import Channel from "@binary/channel";

// Transport
import Snapshot from "@transport/snapshot";

// Security
import Report from "@security/report";

const FLAG_UNCHANGED = 0;
const FLAG_FULL = 1;
const FLAG_DIFF = 2;

interface CachedEntry<K, V> {
    key: K & defined;
    value: V & defined;
    bytes: buffer;
    hash: number;
}

function bufferHash(b: buffer): number {
    const len = buffer.len(b);
    let h = 2166136261;
    for (let i = 0; i < len; i++) {
        h = bit32.bxor(h, buffer.readu8(b, i));
        h = bit32.band(h * 16777619, 0xffffffff);
    }
    return h;
}

function serializeValue<V>(
    scratch: Channel,
    write: (ch: Channel, value: V) => void,
    value: V,
): buffer {
    scratch.offset = 0;
    write(scratch, value);
    return scratch.copyOut(0, scratch.offset);
}

function buffersEqual(a: buffer, b: buffer): boolean {
    const len = buffer.len(a);
    if (len !== buffer.len(b)) return false;
    for (let i = 0; i < len; i++) {
        if (buffer.readu8(a, i) !== buffer.readu8(b, i)) return false;
    }
    return true;
}

function copyMap<K, V>(src: Map<K, V>): Map<K, V> {
    const out = new Map<K, V>();
    src.forEach((v, k) => out.set(k, v));
    return out;
}

function deltaMap<K extends defined, V extends defined>(
    keyCodec: Type.Codec.External<K>,
    valueCodec: Type.Codec.External<V>,
) {
    const id = Snapshot.allocId();
    const internalKey = keyCodec as unknown as Type.Codec.Internal<K>;
    const internalValue = valueCodec as unknown as Type.Codec.Internal<V>;
    const scratch = new Channel();

    return {
        write: (ch, value) => {
            const entries: [K, V][] = [];
            value.forEach((v, k) => entries.push([k, v]));
            entries.sort((a, b) => tostring(a[0]) < tostring(b[0]));

            const cached = ch.deltas.get(id) as Map<string, CachedEntry<K, V>> | undefined;

            if (cached === undefined) {
                ch.u8(FLAG_FULL);
                ch.u16(entries.size());
                const newCache = new Map<string, CachedEntry<K, V>>();
                for (const [k, v] of entries) {
                    internalKey.write(ch, k);
                    const bytes = serializeValue(scratch, internalValue.write, v);
                    const hash = bufferHash(bytes);
                    ch.bytes(bytes, 0, buffer.len(bytes));
                    newCache.set(tostring(k), { key: k, value: v, bytes, hash });
                }
                ch.deltas.set(id, newCache);
                return;
            }

            const incoming = new Map<string, [K, V]>();
            for (const [k, v] of entries) incoming.set(tostring(k), [k, v]);

            const removed: (K & defined)[] = [];
            cached.forEach((entry, ks) => {
                if (!incoming.has(ks)) removed.push(entry.key);
            });

            const upserted: [K, V, buffer, number][] = [];
            for (const [k, v] of entries) {
                const ks = tostring(k);
                const existing = cached.get(ks);

                if (existing !== undefined && existing.value === v) continue;

                const bytes = serializeValue(scratch, internalValue.write, v);
                const hash = bufferHash(bytes);

                if (
                    existing === undefined ||
                    existing.hash !== hash ||
                    !buffersEqual(bytes, existing.bytes)
                ) {
                    upserted.push([k, v, bytes, hash]);
                }
            }

            if (removed.size() === 0 && upserted.size() === 0) {
                ch.u8(FLAG_UNCHANGED);
                return;
            }

            const useDiff = removed.size() + upserted.size() < entries.size();

            if (useDiff) {
                ch.u8(FLAG_DIFF);

                ch.varint(removed.size());
                for (const k of removed) {
                    internalKey.write(ch, k);
                    cached.delete(tostring(k));
                }

                ch.varint(upserted.size());
                for (const [k, v, bytes, hash] of upserted) {
                    internalKey.write(ch, k);
                    ch.bytes(bytes, 0, buffer.len(bytes));
                    cached.set(tostring(k), { key: k, value: v, bytes, hash });
                }
            } else {
                ch.u8(FLAG_FULL);
                ch.u16(entries.size());
                cached.clear();

                for (const [k, v] of entries) {
                    internalKey.write(ch, k);
                    const bytes = serializeValue(scratch, internalValue.write, v);
                    const hash = bufferHash(bytes);
                    ch.bytes(bytes, 0, buffer.len(bytes));
                    cached.set(tostring(k), { key: k, value: v, bytes, hash });
                }
            }
        },

        read: (cursor) => {
            const player = cursor.getPlayer();
            const flag = cursor.u8();

            if (flag === FLAG_UNCHANGED) {
                const cached = Snapshot.getCache(id, player) as Map<K, V> | undefined;
                if (cached === undefined) {
                    Report.log("warn", "DELTA_MAP_UNCHANGED_NO_CACHE", { id });
                    return new Map<K, V>();
                }
                return copyMap(cached);
            }

            if (flag === FLAG_FULL) {
                const len = cursor.u16();
                const result = new Map<K, V>();
                for (let i = 0; i < len; i++) {
                    result.set(internalKey.read(cursor), internalValue.read(cursor));
                }
                Snapshot.setCache(id, result, player);
                return copyMap(result);
            }

            const cached = Snapshot.getCache(id, player) as Map<K, V> | undefined;
            if (cached === undefined) {
                Report.log("warn", "DELTA_MAP_DIFF_NO_CACHE", { id });

                const removedCount = cursor.varint();
                for (let i = 0; i < removedCount; i++) internalKey.read(cursor);

                const upsertCount = cursor.varint();
                for (let i = 0; i < upsertCount; i++) {
                    internalKey.read(cursor);
                    internalValue.read(cursor);
                }

                return new Map();
            }

            const removedCount = cursor.varint();
            for (let i = 0; i < removedCount; i++) {
                cached.delete(internalKey.read(cursor));
            }

            const upsertCount = cursor.varint();
            for (let i = 0; i < upsertCount; i++) {
                cached.set(internalKey.read(cursor), internalValue.read(cursor));
            }

            Snapshot.setCache(id, cached, player);
            return copyMap(cached);
        },

        _default: new Map<K, V>(),
        _delta: true,
    } as Type.Codec.Internal<Map<K, V>> as Type.Codec.External<Map<K, V>>;
}

        for (const key of allKeys) result[key] = internalFields[key]._default;
        return result;
    }

    function copyCache(cached: Record<string, unknown>): Record<string, unknown> {
        const result: Record<string, unknown> = {};
        for (const key of allKeys) result[key] = cached[key];
        return result;
    }

    function readDeltaBitmap(cursor: Type.Cursor): number {
        if (deltaBitmapBytes === 1) return cursor.u8();
        if (deltaBitmapBytes === 2) return cursor.u16();
        return cursor.u32();
    }

    function writeDeltaBitmap(ch: Type.Channel, bitmap: number) {
        if (deltaBitmapBytes === 1) ch.u8(bitmap);
        else if (deltaBitmapBytes === 2) ch.u16(bitmap);
        else ch.u32(bitmap);
    }

    function writeField(ch: Type.Channel, val: Record<string, unknown>, key: string) {
        if (boolKeySet.has(key)) {
            packBoolsKeyed(ch, val, [key], [1], 1);
        } else {
            internalFields[key].write(ch, val[key]);
        }
    }

    function readField(cursor: Type.Cursor, cached: Record<string, unknown>, key: string) {
        if (boolKeySet.has(key)) {
            unpackBoolsKeyed(cursor, cached, [key], [1], 1);
        } else {
            cached[key] = internalFields[key].read(cursor);
        }
    }

    return {
        write: (ch, value) => {
            // Write path: delta cache lives on ch.deltas — no IS_SERVER check
            // or Snapshot.getCurrentPlayer() needed.
            const cached = ch.deltas.get(id) as Record<string, unknown> | undefined;
            const val = value as Record<string, unknown>;

            if (cached === undefined) {
                ch.u8(FLAG_FULL);
                writeFull(ch, val);

                const cache: Record<string, unknown> = {};
                for (const key of allKeys) cache[key] = val[key];
                ch.deltas.set(id, cache);
                return;
            }

            const dirtyIndices: number[] = [];
            for (let i = 0; i < fieldCount; i++) {
                if (cached[allKeys[i]] !== val[allKeys[i]]) dirtyIndices.push(i);
            }

            if (dirtyIndices.size() === 0) {
                ch.u8(FLAG_UNCHANGED);
                return;
            }

            if (dirtyIndices.size() === 1) {
                const i = dirtyIndices[0];
                const key = allKeys[i];
                ch.u8(SINGLE_FIELD_OFFSET + i);
                writeField(ch, val, key);
                cached[key] = val[key];
                return;
            }

            ch.u8(FLAG_MULTI);

            let bitmap = 0;
            for (const i of dirtyIndices) bitmap = bit32.bor(bitmap, bit32.lshift(1, i));
            writeDeltaBitmap(ch, bitmap);

            for (const i of dirtyIndices) {
                const key = allKeys[i];
                writeField(ch, val, key);
                cached[key] = val[key];
            }
        },

        read: (cursor) => {
            const player = cursor.getPlayer();
            const flag = cursor.u8();

            if (flag === FLAG_UNCHANGED) {
                const cached = Snapshot.getCache(id, player) as Record<string, unknown> | undefined;
                if (cached === undefined) {
                    Report.log("warn", "DELTA_STRUCT_UNCHANGED_NO_CACHE", { id });
                    return buildDefault();
                }
                return copyCache(cached) as Type.Codec.InferSchema<T>;
            }

            if (flag === FLAG_FULL) {
                const result = readFull(cursor);
                Snapshot.setCache(id, result, player);
                return copyCache(result) as Type.Codec.InferSchema<T>;
            }

            if (flag === FLAG_MULTI) {
                const cached = Snapshot.getCache(id, player) as Record<string, unknown> | undefined;
                if (cached === undefined) {
                    Report.log("warn", "DELTA_STRUCT_MULTI_NO_CACHE", { id });
                    return buildDefault();
                }
                const bitmap = readDeltaBitmap(cursor);
                for (let i = 0; i < fieldCount; i++) {
                    if (bit32.band(bitmap, bit32.lshift(1, i)) !== 0) {
                        readField(cursor, cached, allKeys[i]);
                    }
                }
                return copyCache(cached) as Type.Codec.InferSchema<T>;
            }

            const fieldIndex = flag - SINGLE_FIELD_OFFSET;
            if (fieldIndex >= fieldCount) {
                Report.log("warn", "DELTA_STRUCT_INVALID_FLAG", { flag, fieldCount });
                return buildDefault();
            }

            const cached = Snapshot.getCache(id, player) as Record<string, unknown> | undefined;
            if (cached === undefined) {
                Report.log("warn", "DELTA_STRUCT_SINGLE_NO_CACHE", { id });
                return buildDefault();
            }

            readField(cursor, cached, allKeys[fieldIndex]);
            return copyCache(cached) as Type.Codec.InferSchema<T>;
        },

        _default: buildDefault() as Type.Codec.InferSchema<T>,
        _delta: true,
    } as Type.Codec.Internal<Type.Codec.InferSchema<T>> as Type.Codec.External<
        Type.Codec.InferSchema<T>
    >;
}

export default deltaStruct;

function encodeZigzag(n: number) {
	return bit32.bxor(bit32.lshift(n, 1), bit32.arshift(n, 31));
}
function decodeZigzag(n: number) {
	return bit32.bxor(bit32.rshift(n, 1), -bit32.band(n, 1));
}

function deltaQuantizedAxis(
	min: number,
	max: number,
	precision: number,
): Type.Codec.External<number> {
	const id = Snapshot.allocId();

	return {
		write: (ch, value) => {
			const q = math.round((math.clamp(value, min, max) - min) / precision);
			const prev = (ch.deltas.get(id) as number | undefined) ?? 0;
			ch.varint(encodeZigzag(q - prev));
			ch.deltas.set(id, q);
		},
		read: (cursor) => {
			const player = cursor.getPlayer();
			const prev = (Snapshot.getCache(id, player) as number | undefined) ?? 0;
			const q = prev + decodeZigzag(cursor.varint());
			Snapshot.setCache(id, q, player);
			return q * precision + min;
		},
		_default: min,
		_delta: true,
	} as Type.Codec.Internal<number> as Type.Codec.External<number>;
}

interface CachedVec3 {
	x: number;
	y: number;
	z: number;
}

function deltaVector3(
	x: Type.Codec.External<number> = Num.f32,
	y: Type.Codec.External<number> = Num.f32,
	z: Type.Codec.External<number> = Num.f32,
): Type.Codec.External<Vector3> {
	const cx = x as unknown as Type.Codec.Internal<number>;
	const cy = y as unknown as Type.Codec.Internal<number>;
	const cz = z as unknown as Type.Codec.Internal<number>;

	if (cx._delta && cy._delta && cz._delta) {
		return {
			write: (ch, value) => {
				cx.write(ch, value.X);
				cy.write(ch, value.Y);
				cz.write(ch, value.Z);
			},
			read: (cursor) => new Vector3(cx.read(cursor), cy.read(cursor), cz.read(cursor)),
			_default: Vector3.zero,
			_delta: true,
		} as Type.Codec.Internal<Vector3> as Type.Codec.External<Vector3>;
	}

	const id = Snapshot.allocId();

	return {
		write: (ch, value) => {
			const cached = ch.deltas.get(id) as CachedVec3 | undefined;
			const xDirty = cached === undefined || cached.x !== value.X;
			const yDirty = cached === undefined || cached.y !== value.Y;
			const zDirty = cached === undefined || cached.z !== value.Z;
			const dirty = (xDirty ? 0b001 : 0) | (yDirty ? 0b010 : 0) | (zDirty ? 0b100 : 0);

			ch.u8(dirty);
			if (dirty === 0) return;

			if (cached === undefined) {
				cx.write(ch, value.X);
				cy.write(ch, value.Y);
				cz.write(ch, value.Z);
				ch.deltas.set(id, { x: value.X, y: value.Y, z: value.Z });
				return;
			}

			if (xDirty) {
				cx.write(ch, value.X);
				cached.x = value.X;
			}
			if (yDirty) {
				cy.write(ch, value.Y);
				cached.y = value.Y;
			}
			if (zDirty) {
				cz.write(ch, value.Z);
				cached.z = value.Z;
			}
		},
		read: (cursor) => {
			const player = cursor.getPlayer();
			const dirty = cursor.u8();

			if (dirty === 0) {
				const cached = Snapshot.getCache(id, player) as Vector3 | undefined;
				if (cached === undefined)
					Report.log("warn", "DELTA_VEC3_UNCHANGED_NO_CACHE", { id });
				return cached ?? Vector3.zero;
			}

			const cached = Snapshot.getCache(id, player) as Vector3 | undefined;
			const nx = bit32.band(dirty, 0b001) !== 0 ? cx.read(cursor) : (cached?.X ?? 0);
			const ny = bit32.band(dirty, 0b010) !== 0 ? cy.read(cursor) : (cached?.Y ?? 0);
			const nz = bit32.band(dirty, 0b100) !== 0 ? cz.read(cursor) : (cached?.Z ?? 0);

			const value = new Vector3(nx, ny, nz);
			Snapshot.setCache(id, value, player);
			return value;
		},
		_default: Vector3.zero,
		_delta: true,
	} as Type.Codec.Internal<Vector3> as Type.Codec.External<Vector3>;
}

function map<K, V>(key: Type.Codec.External<K>, value: Type.Codec.External<V>) {
    const internalKey = key as Type.Codec.Internal<K>;
    const internalValue = value as Type.Codec.Internal<V>;

    return {
        write: (ch, value) => {
            ch.varint(value.size());

            const entries: [K, V][] = [];
            value.forEach((v, k) => entries.push([k, v]));
            entries.sort((a, b) => tostring(a[0]) < tostring(b[0]));

            entries.forEach(([k, v]) => {
                internalKey.write(ch, k);
                internalValue.write(ch, v);
            });
        },
        read: (cursor) => {
            const len = cursor.varint();

            const result = new Map<K, V>();
            for (let i = 0; i < len; i++) {
                const key = internalKey.read(cursor);
                const value = internalValue.read(cursor);
                result.set(key, value);
            }

            return result;
        },
        _default: new Map(),
    } as Type.Codec.Internal<Map<K, V>> as Type.Codec.External<Map<K, V>>;
}

function struct<T extends Record<string, Type.Codec.External<unknown>>>(fields: T) {
    const internalFields = fields as unknown as Record<string, Type.Codec.Internal<unknown>>;
    const allKeys = (Object.keys(internalFields) as string[]).sort();
    const { dataKeys, boolKeys } = splitBoolKeys(internalFields, allKeys);
    const bitmapBytes = getBitmapBytes(boolKeys.size());
    const bitmaskCache = buildBitmaskCache(boolKeys.size());
    const dataLen = dataKeys.size();
    const hasBools = boolKeys.size() > 0;

    const dataCodecs: Type.Codec.Internal<unknown>[] = [];
    const dataKeyArr: string[] = [];

    for (let i = 0; i < dataLen; i++) {
        dataKeyArr[i] = dataKeys[i];
        dataCodecs[i] = internalFields[dataKeys[i]];
    }

    let allDirect = dataLen > 0;
    let fixedSize = 0;

    for (let i = 0; i < dataLen; i++) {
        const c = dataCodecs[i];
        if (c._size !== undefined && c._directWrite !== undefined && c._directRead !== undefined) {
            fixedSize += c._size;
        } else {
            allDirect = false;
            break;
        }
    }

    const boolBytes = getBitmapBytes(boolKeys.size());
    const totalFixed = fixedSize + boolBytes;

    const defaultResult: Record<string, unknown> = {};
    for (const key of allKeys) defaultResult[key] = internalFields[key]._default;

    if (allDirect) {
        const offsets: number[] = [];
        const directWrites: ((buf: buffer, off: number, value: unknown) => void)[] = [];
        const directReads: ((buf: buffer, off: number) => unknown)[] = [];

        let runningOffset = 0;
        for (let i = 0; i < dataLen; i++) {
            offsets[i] = runningOffset;
            directWrites[i] = dataCodecs[i]._directWrite!;
            directReads[i] = dataCodecs[i]._directRead!;
            runningOffset += dataCodecs[i]._size!;
        }

        const directWrite = (buf: buffer, off: number, value: unknown) => {
            const val = value as Record<string, unknown>;
            for (let i = 0; i < dataLen; i++) {
                directWrites[i](buf, off + offsets[i], val[dataKeyArr[i]]);
            }
            if (hasBools) {
                let bitmap = 0;
                let bi = 0;
                for (const key of boolKeys) {
                    if (val[key] === true) bitmap = bit32.bor(bitmap, bitmaskCache[bi]);
                    bi++;
                }
                if (boolBytes === 1) buffer.writeu8(buf, off + fixedSize, bitmap);
                else if (boolBytes === 2) buffer.writeu16(buf, off + fixedSize, bitmap);
                else buffer.writeu32(buf, off + fixedSize, bitmap);
            }
        };

        const directRead = (buf: buffer, off: number) => {
            const result: Record<string, unknown> = {};
            for (let i = 0; i < dataLen; i++) {
                result[dataKeyArr[i]] = directReads[i](buf, off + offsets[i]);
            }
            if (hasBools) {
                let bitmap: number;
                if (boolBytes === 1) bitmap = buffer.readu8(buf, off + fixedSize);
                else if (boolBytes === 2) bitmap = buffer.readu16(buf, off + fixedSize);
                else bitmap = buffer.readu32(buf, off + fixedSize);

                let bi = 0;
                for (const key of boolKeys) {
                    result[key] = bit32.band(bitmap, bitmaskCache[bi]) !== 0;
                    bi++;
                }
            }
            return result;
        };

        return {
            write: (ch, value) => {
                // Direct path: write all fields in one contiguous block via
                // raw buffer access. Channel guarantees contiguous memory.
                ch.ensureSpace(totalFixed);
                directWrite(ch.buf, ch.offset, value);
                ch.offset += totalFixed;
            },
            read: (cursor) => {
                cursor.ensureRemaining(totalFixed);
                const result = directRead(cursor.buf, cursor.offset);
                cursor.offset += totalFixed;
                return result;
            },
            _directWrite: directWrite,
            _directRead: directRead,
            _default: defaultResult,
            _size: totalFixed,
        } as Type.Codec.Internal<Type.Codec.InferSchema<T>> as Type.Codec.External<
            Type.Codec.InferSchema<T>
        >;
    }

    return {
        write: (ch, value) => {
            const val = value as Record<string, unknown>;
            for (let i = 0; i < dataLen; i++) {
                dataCodecs[i].write(ch, val[dataKeyArr[i]]);
            }
            if (hasBools) packBoolsKeyed(ch, val, boolKeys, bitmaskCache, bitmapBytes);
        },
        read: (cursor) => {
            const result: Record<string, unknown> = {};
            for (let i = 0; i < dataLen; i++) {
                result[dataKeyArr[i]] = dataCodecs[i].read(cursor);
            }
            if (hasBools) unpackBoolsKeyed(cursor, result, boolKeys, bitmaskCache, bitmapBytes);
            return result;
        },
        _default: defaultResult,
    } as Type.Codec.Internal<Type.Codec.InferSchema<T>> as Type.Codec.External<
        Type.Codec.InferSchema<T>
    >;
}

function union<V extends Record<string, Type.Codec.External<unknown>>>(variants: V) {
    const keys = (Object.keys(variants) as string[]).sort();
    const internalVariants = variants as unknown as Record<string, Type.Codec.Internal<unknown>>;

    if (keys.size() > Constant.U8_MAX) {
        Report.log("warn", "CODEC_UNION_TOO_MANY_VARIANTS", { count: keys.size() });
    }

    const keyToIndex = new Map<string, number>();
    keys.forEach((key, i) => keyToIndex.set(key, i));

    return {
        write: (ch, value) => {
            const _value = value as { type: string } & Record<string, unknown>;
            const _type = _value.type;
            const index = keyToIndex.get(_type);

            if (index === undefined) {
                Report.log("warn", "CODEC_UNION_UNKNOWN_VARIANT", { type: _type });
                ch.u8(0);
                return;
            }

            ch.u8(index);

            const data: Record<string, unknown> = {};
            for (const [k, v] of Object.entries(_value)) {
                if (k !== "type") data[k] = v;
            }

            internalVariants[_type].write(ch, data);
        },
        read: (cursor) => {
            const index = cursor.u8();

            if (index >= keys.size()) {
                Report.log("warn", "CODEC_UNION_INVALID_INDEX", { index, max: keys.size() });
                return { type: keys[0] };
            }

            const _type = keys[index];
            const data = internalVariants[_type].read(cursor) as Record<string, unknown>;

            data["type"] = _type;
            return data;
        },
        _default: { type: keys[0] },
    } as Type.Codec.Internal<Type.Codec.Union<V>> as Type.Codec.External<Type.Codec.Union<V>>;
}

import Constant from "@constant";
import * as Type from "@type";

// Security
import Report from "@security/report";

function float(min: number, max: number, precision?: number): Type.Codec.External<number> {
    if (precision === undefined) {
        return {
            write: (ch, value) => ch.f16(value),
            read: (cursor) => cursor.f16(),
            _default: min,
        } as Type.Codec.Internal<number> as Type.Codec.External<number>;
    }

    const steps = math.floor((max - min) / precision);

    let write: (ch: Type.Channel, value: number) => void;
    let read: (cursor: Type.Cursor) => number;

    if (steps <= Constant.U8_MAX) {
        write = (ch, value) => ch.u8(math.round((value - min) / precision));
        read = (cursor) => cursor.u8() * precision + min;
    } else if (steps <= Constant.U16_MAX) {
        write = (ch, value) => ch.u16(math.round((value - min) / precision));
        read = (cursor) => cursor.u16() * precision + min;
    } else if (steps <= Constant.U24_MAX) {
        write = (ch, value) => ch.u24(math.round((value - min) / precision!));
        read = (cursor) => cursor.u24() * precision! + min;
    } else if (steps <= Constant.U32_MAX) {
        write = (ch, value) => ch.u32(math.round((value - min) / precision));
        read = (cursor) => cursor.u32() * precision + min;
    } else {
        Report.log("warn", "CODEC_FLOAT_STEPS_EXCEEDED", { min, max, precision, steps });

        write = (ch, value) => ch.f32(value);
        read = (cursor) => cursor.f32();
    }

    return {
        write,
        read,

        _default: min,
    } as Type.Codec.Internal<number> as Type.Codec.External<number>;
}

export default float;

function int(min: number, max: number): Type.Codec.External<number> {
    const range = max - min;

    let write: (ch: Type.Channel, value: number) => void;
    let read: (cursor: Type.Cursor) => number;
    let directWrite: (buf: buffer, off: number, value: number) => void;
    let directRead: (buf: buffer, off: number) => number;

    let size: number | undefined;

    if (range <= Constant.U8_MAX) {
        write = (ch, value) => ch.u8(value - min);
        read = (cursor) => cursor.u8() + min;
        directWrite = (b, o, v) => buffer.writeu8(b, o, v - min);
        directRead = (b, o) => buffer.readu8(b, o) + min;

        size = 1;
    } else if (range <= Constant.U16_MAX) {
        write = (ch, value) => ch.u16(value - min);
        read = (cursor) => cursor.u16() + min;
        directWrite = (b, o, v) => buffer.writeu16(b, o, v - min);
        directRead = (b, o) => buffer.readu16(b, o) + min;

        size = 2;
    } else if (range <= Constant.U24_MAX) {
        write = (ch, value) => ch.varint(value - min);
        read = (cursor) => cursor.varint() + min;
        size = undefined;
    } else {
        write = (ch, value) => ch.varint(value - min);
        read = (cursor) => cursor.varint() + min;
        size = undefined;
    }

    return {
        write,
        read,
        _directWrite: directWrite!,
        _directRead: directRead!,

        _default: min,

        _size: size,
    } as Type.Codec.Internal<number> as Type.Codec.External<number>;
}

function literal<T extends readonly (string | number | boolean)[]>(
    ...values: T
): Type.Codec.External<T[number]> {
    if (values.size() > Constant.U8_MAX) {
        Report.log("warn", "CODEC_LITERAL_TOO_MANY_VALUES", { count: values.size() });
    }

    return {
        write: (ch, value) => {
            const index = (values as unknown as defined[]).indexOf(value as defined);
            if (index === -1) {
                Report.log("warn", "CODEC_LITERAL_UNKNOWN_VALUE", {
                    value: tostring(value as defined),
                });
                ch.u8(0);
                return;
            }
            ch.u8(index);
        },
        read: (cursor) => {
            const index = cursor.u8();
            if (index >= values.size()) {
                Report.log("warn", "CODEC_LITERAL_INVALID_INDEX", { index, max: values.size() });
                return values[0];
            }
            return values[index];
        },
        _default: values[0],
    } as Type.Codec.Internal<T[number]> as Type.Codec.External<T[number]>;
}

function encodeZigzag(n: number): number {
    return bit32.bxor(bit32.lshift(n, 1), bit32.arshift(n, 31));
}
function decodeZigzag(n: number): number {
    return bit32.bxor(bit32.rshift(n, 1), -bit32.band(n, 1));
}

const zint = {
    write: (ch, value) => {
        let n = encodeZigzag(value);

        while (n > 0x7f) {
            ch.u8(bit32.bor(bit32.band(n, 0x7f), 0x80));
            n = bit32.rshift(n, 7);
        }
        ch.u8(n);
    },
    read: (cursor) => {
        let n = 0;
        let shift = 0;

        while (true) {
            const byte = cursor.u8();
            n = bit32.bor(n, bit32.lshift(bit32.band(byte, 0x7f), shift));
            shift += 7;

            if (bit32.band(byte, 0x80) === 0) break;
        }

        return decodeZigzag(n);
    },
    _default: 0,
} as Type.Codec.Internal<number> as Type.Codec.External<number>;

function extractQuaternion(cf: CFrame): [number, number, number, number] {
	const [_, __, ___, r00, r01, r02, r10, r11, r12, r20, r21, r22] = cf.GetComponents();

	const trace = r00 + r11 + r22;
	let qx: number, qy: number, qz: number, qw: number;

	if (trace > 0) {
		const s = 0.5 / math.sqrt(trace + 1);
		qw = 0.25 / s;
		qx = (r21 - r12) * s;
		qy = (r02 - r20) * s;
		qz = (r10 - r01) * s;
	} else if (r00 > r11 && r00 > r22) {
		const s = 2 * math.sqrt(1 + r00 - r11 - r22);
		qw = (r21 - r12) / s;
		qx = 0.25 * s;
		qy = (r01 + r10) / s;
		qz = (r02 + r20) / s;
	} else if (r11 > r22) {
		const s = 2 * math.sqrt(1 + r11 - r00 - r22);
		qw = (r02 - r20) / s;
		qx = (r01 + r10) / s;
		qy = 0.25 * s;
		qz = (r12 + r21) / s;
	} else {
		const s = 2 * math.sqrt(1 + r22 - r00 - r11);
		qw = (r10 - r01) / s;
		qx = (r02 + r20) / s;
		qy = (r12 + r21) / s;
		qz = 0.25 * s;
	}

	return [qx, qy, qz, qw];
}

function cframe(
	minPos?: number | Type.Codec.Vector3,
	maxPos?: number | Type.Codec.External<number>,
	precision?: number,
): Type.Codec.External<CFrame> {
	let posCodec: Type.Codec.External<Vector3>;

	if (
		typeOf(minPos) === "number" &&
		typeOf(maxPos) === "number" &&
		typeOf(precision) === "number"
	) {
		const mn = minPos as number;
		const mx = maxPos as number;
		const pr = precision as number;
		posCodec = (
			vector3 as (min: number, max: number, precision: number) => Type.Codec.External<Vector3>
		)(mn, mx, pr);
	} else {
		posCodec = ((minPos as Type.Codec.Vector3 | undefined) ??
			vector3) as Type.Codec.External<Vector3>;
	}

	const internalPosition = posCodec as Type.Codec.Internal<Vector3>;

	return {
		write: (ch, value) => {
			internalPosition.write(ch, value.Position);

			const [qx, qy, qz, qw] = extractQuaternion(value);
			const components = [qx, qy, qz, qw];

			let largestIndex = 0;
			let largestValue = math.abs(components[0]);
			for (let i = 1; i < 4; i++) {
				if (math.abs(components[i]) > largestValue) {
					largestValue = math.abs(components[i]);
					largestIndex = i;
				}
			}

			const sign = components[largestIndex] < 0 ? -1 : 1;

			let packed = largestIndex;
			let shift = 2;
			for (let i = 0; i < 4; i++) {
				if (i === largestIndex) continue;
				const v = components[i] * sign;
				const q = math.clamp(
					math.round((v + Constant.QUAT_OFFSET) * Constant.QUAT_SCALE),
					0,
					1023,
				);
				packed = bit32.bor(packed, bit32.lshift(q, shift));
				shift += 10;
			}

			ch.u32(packed);
		},

		read: (cursor) => {
			const pos = internalPosition.read(cursor);

			const packed = cursor.u32();
			const largestIndex = bit32.band(packed, 0x3);

			const values: number[] = [];
			for (let i = 0; i < 3; i++) {
				const q = bit32.band(bit32.rshift(packed, 2 + i * 10), 0x3ff);
				values[i] = q / Constant.QUAT_SCALE - Constant.QUAT_OFFSET;
			}

			const sumSq = values[0] ** 2 + values[1] ** 2 + values[2] ** 2;
			const largest = math.sqrt(math.max(0, 1 - sumSq));

			const q: number[] = [];
			let vi = 0;
			for (let i = 0; i < 4; i++) {
				q[i] = i === largestIndex ? largest : values[vi++];
			}

			return new CFrame(
				pos.X,
				pos.Y,
				pos.Z,
				1 - 2 * (q[1] ** 2 + q[2] ** 2),
				2 * (q[0] * q[1] - q[2] * q[3]),
				2 * (q[0] * q[2] + q[1] * q[3]),
				2 * (q[0] * q[1] + q[2] * q[3]),
				1 - 2 * (q[0] ** 2 + q[2] ** 2),
				2 * (q[1] * q[2] - q[0] * q[3]),
				2 * (q[0] * q[2] - q[1] * q[3]),
				2 * (q[1] * q[2] + q[0] * q[3]),
				1 - 2 * (q[0] ** 2 + q[1] ** 2),
			);
		},
		_default: new CFrame(),
	} as Type.Codec.Internal<CFrame> as Type.Codec.External<CFrame>;
}

	},

import int from "@codec/primitive/int";

function quantizedAxis(min: number, max: number, precision: number): Type.Codec.External<number> {
    const steps = math.round((max - min) / precision);
    const codec = int(0, steps) as unknown as Type.Codec.Internal<number>;
    return {
        write: (ch, value) =>
            codec.write(ch, math.round((math.clamp(value, min, max) - min) / precision)),
        read: (cursor) => codec.read(cursor) * precision + min,
        _default: min,
        _size: codec._size,
    } as Type.Codec.Internal<number> as Type.Codec.External<number>;
}

function vector3(
    x: Type.Codec.External<number>,
    y: Type.Codec.External<number>,
    z: Type.Codec.External<number>,

    const internalY = y as Type.Codec.Internal<number>;

        write: (ch, value) => {
            internalX.write(ch, value.X);
            internalY.write(ch, value.Y);
            internalZ.write(ch, value.Z);

        read: (cursor) =>
            new Vector3(internalX.read(cursor), internalY.read(cursor), internalZ.read(cursor)),
        _default: Vector3.zero,
    } as Type.Codec.Internal<Vector3> as Type.Codec.External<Vector3>;

//!optimize 2

// Binary
import Channel from "@binary/channel";

// Simple array-backed LIFO stack — no dependency on Array.pop() for
// roblox-ts compatibility.
const backing: Channel[] = [];
let poolSize = 0;

// Acquire a reset Channel from the pool, or allocate a fresh one.
// The returned channel has offset=0, empty deltas, and prevHash=0.
export function acquire(): Channel {
    if (poolSize > 0) {
        poolSize--;
        const ch = backing[poolSize];
        ch.offset = 0;
        ch.deltas.clear();
        ch.prevHash = 0;
        return ch;
    }
    return new Channel();
}

// Return a channel to the pool. Caller must not use the channel after release.
// Deltas are cleared so the next acquire() starts with a clean slate.
export function release(ch: Channel): void {
    ch.offset = 0;
    ch.deltas.clear();
    ch.prevHash = 0;
    backing[poolSize] = ch;
    poolSize++;
}

const Pool = { acquire, release };
export default Pool;

import Constant from "@constant";

type Severity = "debug" | "warn" | "kick" | "fatal";

type LogContext = {
    player?: Player;
    [key: string]: unknown;
};

const CODE_WIDTH = 22;
const PLAYER_WIDTH = 14;
const PREFIX = Constant.LIB_NAME;

let debugEnabled = false;

function pad(s: string, width: number): string {
    let result = s;
    while (result.size() < width) result += " ";
    return result;
}

function formatContext(context: LogContext): string {
    const parts: string[] = [];
    for (const [key, value] of pairs(context)) {
        if (key === "player") continue;
        parts.push(`${key}=${tostring(value)}`);
    }
    return parts.join(" ");
}

function log(severity: Severity, code: string, context: LogContext = {}) {
    if (severity === "debug" && !debugEnabled) return;

    const playerName = context.player !== undefined ? `[${context.player.Name}]` : "";
    const playerSlot = pad(playerName, PLAYER_WIDTH);
    const details = formatContext(context);

    const tag = severity === "debug" ? "·" : `[${PREFIX}]`;
    const paddedCode = pad(code, CODE_WIDTH);
    const detailsPart = details !== "" ? ` · ${details}` : "";

    const message = `${tag} ${playerSlot} ${paddedCode}${detailsPart}`;

    if (severity === "warn" || severity === "kick") {
        warn(message);
    } else {
        print(message);
    }

    if (severity === "kick" && context.player !== undefined) {
        context.player.Kick(`Network violation: ${code}`);
    }

    if (severity === "fatal") {
        error(message);
    }
}

function setDebug(enabled: boolean) {
    debugEnabled = enabled;
}

const Report = {
    log,
    setDebug,
};

// Internal
import { IS_SERVER } from "@environment";
import Constant from "@constant";

// Binary
import Cursor from "@binary/cursor";

// Transport
import Lifecycle from "./lifecycle";
import Registry from "./registry";

// Security
import Report from "@security/report";

function handle(buf: buffer, player: Player) {
    if (buffer.len(buf) > Constant.MAX_PAYLOAD) {
        Report.log("kick", "INBOUND_PAYLOAD_TOO_LARGE", { player, size: buffer.len(buf) });
        return;
    }

    if (buffer.len(buf) === 0) {
        Report.log("kick", "INBOUND_EMPTY_BUFFER", { player });
        return;
    }

    parse(buf, player);
}

function parse(buf: buffer, player: Player) {
    const cursor = new Cursor(buf, IS_SERVER ? player : undefined);

    while (cursor.remaining() >= 1) {
        const id = cursor.varint();

        if (id === Constant.HANDSHAKE_PACKET_ID) {
            if (IS_SERVER) {
                Report.log("debug", "INBOUND_HANDSHAKE", { player });
                Lifecycle.markReady(player);
            }
            continue;
        }

        if (IS_SERVER && !Lifecycle.isReady(player)) {
            Report.log("kick", "INBOUND_NOT_READY", { player });
            return;
        }

        const def = Registry.getById(id);
        if (def === undefined) {
            Report.log("warn", "INBOUND_UNKNOWN_PACKET", {
                player,
                id,
                bufLen: buffer.len(cursor.buf),
                offset: cursor.offset,
                b2: buffer.readu8(cursor.buf, 2),
                b3: buffer.readu8(cursor.buf, 3),
                b4: buffer.readu8(cursor.buf, 4),
                b5: buffer.readu8(cursor.buf, 5),
            });
            return;
        }

        if (def.handler === undefined) {
            Report.log("warn", "INBOUND_NO_HANDLER", { player, id, name: def.name });
        }

        const data = def.codec !== undefined ? def.codec.read(cursor) : undefined;
        if (def.handler !== undefined) {
            def.handler(player, data);
        }
    }

    if (cursor.remaining() > 0) {
        Report.log("kick", "INBOUND_LEFTOVER_BYTES", { player, remaining: cursor.remaining() });
    }
}

const Inbound = {
    handle,
};

// Utility

// Security
import Report from "@security/report";
import Constant from "@constant";

let started = false;

const reliableChannels = new Map<Player, Channel>();
const unreliableChannels = new Map<Player, Channel>();
const scratch = new Channel();

let clientChannel: Channel;

const flushQueue: Type.FlushEntry[] = [];
const heldBuffers: Type.HeldBuffer[] = [];

function hashChannel(ch: Channel): number {
    let h = 2166136261;
    const buf = ch.buf;
    const len = ch.offset;
    for (let i = 0; i < len; i++) {
        h = bit32.bxor(h, buffer.readu8(buf, i));
        h = (h * 16777619) % 4294967296;
    }
    return h;
}

function getPlayerChannel(map: Map<Player, Channel>, player: Player): Channel {
    if (!map.has(player)) {
        map.set(player, Pool.acquire());
    }
    return map.get(player)!;
}

function send<T>(
    id: number,
    codec: Type.Codec.Internal<T> | undefined,
    data: T | undefined,
    target: Type.Target | undefined,
    unreliable: boolean,
) {
    if (!started) {
        Report.log("warn", "OUTBOUND_NOT_STARTED", { id });
        return;
    }

    if (!IS_SERVER && id === Constant.HANDSHAKE_PACKET_ID) {
        const ch = Pool.acquire();
        ch.varint(id);

        const buf = ch.copyOut(0, ch.offset);
        Pool.release(ch);

        Bridge.fireReliable(buf);

        return;
    }
    Report.log("debug", "OUTBOUND_SEND_CALLED", { id, targetIsUndefined: target === undefined });

    flushQueue.push({
        id,
        target,
        unreliable,
        codec: codec as Type.Codec.Internal<unknown> | undefined,
        data: data as unknown,
    });
}

function flush() {
    if (flushQueue.size() === 0) return;

    if (IS_SERVER) {
        for (const entry of flushQueue) {
            const players = resolveTarget(entry.target);

            if (players.size() === 0) {
                Report.log("debug", "OUTBOUND_NO_RECIPIENTS", { id: entry.id });
            }

            if (entry.codec?._delta) {
                for (const player of players) {
                    const ch = getPlayerChannel(
                        entry.unreliable ? unreliableChannels : reliableChannels,
                        player,
                    );
                    const beforeOffset = ch.offset;
                    ch.varint(entry.id);
                    const afterId = ch.offset;
                    entry.codec.write(ch, entry.data);

                    const written = ch.offset - afterId;
                    if (written === 1) {
                        ch.offset = beforeOffset;
                    }

                    Report.log("debug", "OUTBOUND_DELTA_WRITTEN", { player, id: entry.id });
                }
            } else {
                scratch.offset = 0;
                scratch.varint(entry.id);
                if (entry.codec !== undefined) {
                    entry.codec.write(scratch, entry.data);
                }
                const encodedLen = scratch.offset;

                for (const player of players) {
                    const ch = getPlayerChannel(
                        entry.unreliable ? unreliableChannels : reliableChannels,
                        player,
                    );
                    ch.bytes(scratch.buf, 0, encodedLen);
                }
            }
        }

        for (const [player, ch] of reliableChannels) {
            const used = ch.offset;
            if (used === 0) continue;

            const hash = hashChannel(ch);
            if (hash === ch.prevHash) {
                ch.offset = 0;
                Report.log("debug", "OUTBOUND_FRAME_UNCHANGED", { player });
                continue;
            }
            ch.prevHash = hash;

            const buf = ch.copyOut(0, used);
            ch.offset = 0;

            if (!Lifecycle.isReady(player)) {
                holdForPlayer(player, buf, false);
            } else {
                fireToPlayer(player, buf, false);
            }
        }

        for (const [player, ch] of unreliableChannels) {
            const used = ch.offset;
            if (used === 0) continue;

            const buf = ch.copyOut(0, used);
            ch.offset = 0;

            if (Lifecycle.isReady(player)) {
                fireToPlayer(player, buf, true);
            }
        }
    } else {
        for (const entry of flushQueue) {
            if (entry.codec?._delta) {
                const beforeOffset = clientChannel.offset;
                clientChannel.varint(entry.id);

                const afterId = clientChannel.offset;
                entry.codec.write(clientChannel, entry.data);

                if (clientChannel.offset - afterId === 1) {
                    clientChannel.offset = beforeOffset;
                }
            } else {
                clientChannel.varint(entry.id);

                if (entry.codec !== undefined) {
                    entry.codec.write(clientChannel, entry.data);
                }
            }
        }

        const used = clientChannel.offset;
        if (used > 0) {
            const buf = clientChannel.copyOut(0, used);
            clientChannel.offset = 0;
            Bridge.fireReliable(buf);
        }
    }

    flushQueue.clear();
}

function holdForPlayer(player: Player, buf: buffer, unreliable: boolean) {
    heldBuffers.push({ player, buf, unreliable });
    Report.log("debug", "OUTBOUND_HELD", { player, raw: buffer.len(buf) });
}

function fireToPlayer(player: Player, buf: buffer, unreliable: boolean) {
    if (unreliable) {
        Bridge.fireUnreliable(buf, player);
    } else {
        Bridge.fireReliable(buf, player);
    }
    Report.log("debug", "OUTBOUND_FIRED", { player, raw: buffer.len(buf), unreliable });
}

function releasePlayer(player: Player) {
    const stillHeld: Type.HeldBuffer[] = [];
    let released = 0;

    for (const held of heldBuffers) {
        if (held.player === player) {
            fireToPlayer(held.player, held.buf, held.unreliable);
            released++;
        } else {
            stillHeld.push(held);
        }
    }

    if (released > 0) {
        Report.log("debug", "OUTBOUND_RELEASED", { player, count: released });
    }

    heldBuffers.clear();
    for (const h of stillHeld) heldBuffers.push(h);
}

function cleanupPlayer(player: Player) {
    const relCh = reliableChannels.get(player);
    if (relCh !== undefined) {
        reliableChannels.delete(player);
        Pool.release(relCh);
    }

    const unrelCh = unreliableChannels.get(player);
    if (unrelCh !== undefined) {
        unreliableChannels.delete(player);
        Pool.release(unrelCh);
    }

    Snapshot.clearPlayer(player);

    const stillHeld: Type.HeldBuffer[] = [];
    let dropped = 0;

    for (const held of heldBuffers) {
        if (held.player !== player) {
            stillHeld.push(held);
        } else {
            dropped++;
        }
    }

    if (dropped > 0) {
        Report.log("debug", "OUTBOUND_DROPPED", { player, count: dropped });
    }

    heldBuffers.clear();
    for (const h of stillHeld) heldBuffers.push(h);
}

function start() {
    if (!IS_SERVER) {
        clientChannel = Pool.acquire();
    }

    started = true;

    Lifecycle.onReady((player) => releasePlayer(player));
    Lifecycle.onLeave((player) => cleanupPlayer(player));
}

const Outbound = {
    start,
    send,
    flush,
};

export default Outbound;

// Internal
import * as Type from "@type";

// Security
import Report from "@security/report";

export interface Definition {
    id: number;
    name: string;
    codec: Type.Codec.Internal<unknown> | undefined;
    handler: ((player: Player, data: unknown) => void) | undefined;
    unreliable: boolean;
}

const byName = new Map<string, Definition>();
const byId = new Map<number, Definition>();
const pending: Definition[] = [];

let finalized = false;

function register(
    name: string,
    codec: Type.Codec.Internal<unknown> | undefined,
    unreliable: boolean,
): Definition {
    if (finalized) {
        Report.log("fatal", "REGISTRY_ALREADY_FINALIZED", { name });
    }

    if (byName.has(name)) {
        Report.log("fatal", "REGISTRY_DUPLICATE_NAME", { name });
    }

    const def: Definition = {
        id: 0,
        name,
        codec,
        handler: undefined,
        unreliable,
    };

    byName.set(name, def);
    pending.push(def);

    return def;
}

function finalize() {
    if (finalized) return;
    finalized = true;

    pending.sort((a, b) => a.name < b.name);

    let nextId = 1;
    for (const def of pending) {
        def.id = nextId++;
        byId.set(def.id, def);
    }

    pending.clear();
}

function setHandler(name: string, handler: (player: Player, data: unknown) => void) {
    const def = byName.get(name);
    if (def === undefined) {
        Report.log("warn", "REGISTRY_HANDLER_FOR_UNKNOWN", { name });
        return;
    }
    def.handler = handler;
}

function getById(id: number): Definition | undefined {
    return byId.get(id);
}

function getByName(name: string): Definition | undefined {
    return byName.get(name);
}

const Registry = {
    register,
    finalize,
    setHandler,
    getById,
    getByName,
};

export default Registry;

import { IS_SERVER } from "@environment";

// Security
import Report from "@security/report";

// ─── ID Allocation ────────────────────────────────────────────────────────────
// Delta codecs call allocId() once at definition time to claim a stable slot
// in the per-channel deltas Map (write path) and in the read-path caches below.

let nextId = 1;

export function allocId(): number {
    return nextId++;
}

// ─── Read-path delta cache ────────────────────────────────────────────────────
// Used exclusively by the INBOUND decode path (cursor-based reads).
// The WRITE path no longer uses Snapshot — each player's Channel owns its
// own `deltas: Map<number, unknown>` instead.

const clientCache = new Map<number, unknown>();
const serverCache = new Map<number, Map<Player, unknown>>();

export function getCache(id: number, player?: Player): unknown | undefined {
    if (IS_SERVER) {
        if (player === undefined) {
            Report.log("warn", "BASELINE_SERVER_NO_PLAYER", { id });
            return undefined;
        }
        return serverCache.get(id)?.get(player);
    }
    return clientCache.get(id);
}

export function setCache(id: number, value: unknown, player?: Player): void {
    if (IS_SERVER) {
        if (player === undefined) {
            Report.log("warn", "BASELINE_SERVER_NO_PLAYER", { id });
            return;
        }
        if (!serverCache.has(id)) serverCache.set(id, new Map());
        serverCache.get(id)!.set(player, value);
    } else {
        clientCache.set(id, value);
    }
}

export function clearPlayer(player: Player): void {
    for (const [_, playerMap] of serverCache) {
        playerMap.delete(player);
    }
}

const Snapshot = {
    allocId,
    getCache,
    setCache,
    clearPlayer,
};

export default Snapshot;

import * as Type from "@type";

export function packBoolsKeyed(
    ch: Type.Channel,
    value: Record<string, unknown>,
    boolKeys: string[],
    bitmaskCache: number[],
    bitmapBytes: number,
) {
    let bitmap = 0;
    let i = 0;
    for (const key of boolKeys) {
        if (value[key] === true) bitmap = bit32.bor(bitmap, bitmaskCache[i]);
        i++;
    }
    if (bitmapBytes === 1) ch.u8(bitmap);
    else if (bitmapBytes === 2) ch.u16(bitmap);
    else ch.u32(bitmap);
}

export function unpackBoolsKeyed(
    cursor: Type.Cursor,
    result: Record<string, unknown>,
    boolKeys: string[],
    bitmaskCache: number[],
    bitmapBytes: number,
) {
    let bitmap: number;
    if (bitmapBytes === 1) bitmap = cursor.u8();
    else if (bitmapBytes === 2) bitmap = cursor.u16();
    else bitmap = cursor.u32();

    let i = 0;
    for (const key of boolKeys) {
        result[key] = bit32.band(bitmap, bitmaskCache[i]) !== 0;
        i++;
    }
}

export function getBitmapBytes(boolCount: number): number {
    return boolCount === 0 ? 0 : boolCount <= 8 ? 1 : boolCount <= 16 ? 2 : 4;
}

export function buildBitmaskCache(count: number): number[] {
    const cache: number[] = [];
    for (let i = 0; i < count; i++) cache[i] = bit32.lshift(1, i);
    return cache;
}

export function splitBoolKeys(
    internalFields: Record<string, Type.Codec.Internal<unknown>>,
    allKeys: string[],
): { dataKeys: string[]; boolKeys: string[] } {
    const dataKeys: string[] = [];
    const boolKeys: string[] = [];
    for (const key of allKeys) {
        if (typeOf(internalFields[key]._default) === "boolean") {
            boolKeys.push(key);
        } else {
            dataKeys.push(key);
        }
    }
    return { dataKeys, boolKeys };

// Package
import { Players } from "@rbxts/services";

// Internal
import * as Type from "@type";

function isPlayer(value: unknown): value is Player {
    return typeIs(value, "Instance") && value.IsA("Player");
}

function isExcludeTuple(target: Type.Target): target is ["Exclude", Player | Player[]] {
    return typeIs(target, "table") && (target as defined[])[0] === "Exclude";
}

function isPlayerArray(value: Player | Player[]): value is Player[] {
    return typeIs(value, "table");
}

function resolveTarget(target: Type.Target | undefined): Player[] {
    if (target === undefined) return Players.GetPlayers();
    if (isPlayer(target)) return [target];

    if (isExcludeTuple(target)) {
        const excluded = target[1];
        const excludedArray = isPlayerArray(excluded) ? excluded : [excluded];
        const excludedSet = new Set<Player>(excludedArray);

        const result: Player[] = [];
        for (const player of Players.GetPlayers()) {
            if (!excludedSet.has(player)) {
                result.push(player);
            }
        }
        return result;
    }

    return target as Player[];
}

export default resolveTarget;
